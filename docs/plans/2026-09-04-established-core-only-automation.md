# Established Core-only Automation Implementation Plan

> **For the executing agent:** Implement this plan task-by-task by loading the
> `executing-plans` and `tdd` skills. Steps use checkbox (`- [ ]`) syntax for
> tracking. Each task follows Red → Green → Refactor inline.

**Goal:** Let established Core-only repositories create a verified project manifest, reconcile Core automation, activate canonical hooks, and commit without invoking a stack adapter.

**Architecture:** Implement accepted ADR-0105 by centralizing project-manifest rendering and validation, making adapter automation optional only when adapter evidence is genuinely absent, and adding the established manifest as a Core provider in the existing journaled automation transaction. Preserve strict discovery failures, ownership conflicts, exact Git preconditions, separate hook approval, and closed bounded diagnostics.

**Tech Stack:** Node.js CommonJS, `node:test`, filesystem no-follow reads, existing `prism-tool` transaction helpers, Bash contract tests, Markdown.

**Originating issue:** #515

## Global constraints

- Follow `adr/0105-established-project-manifest-and-core-only-composition.md` and the continuing constraints in ADR-0078, ADR-0084, and ADR-0100.
- A genuine missing adapter selects `CORE_ONLY`; malformed, ambiguous, escaping, symlinked, mismatched, or incomplete adapter evidence remains `NO-GO`.
- Schema-one Blank and Template project manifests remain valid and are not rewritten solely because the repository is now established.
- New established manifests use schema version two and exact source `{ "mode": "ESTABLISHED", "evidence": null }`.
- `.prism/project.json` is created or updated only inside the approved automation transaction and must verify before hook reconciliation.
- Keep project inputs closed, UTF-8, bounded, canonical, no-follow, and project-contained; never expose raw exceptions, payloads, subprocess output, or arbitrary paths in diagnostics.
- Preserve arbitrary unowned automation and manifest files byte-for-byte as conflicts.
- Use no new runtime dependency and add no public test/provider injection control.
- Every new or modified JavaScript source keeps the RCS header and final vim modeline.
- Run Node tests through the repository's declared npm script for final verification; focused Red/Green cycles may use `node --test` with exact test files and name patterns.

// $KYAULabs: project-manifest.js kyau@aura.kyaulabs 2026/09/04 -0700 Exp $

'use strict';

const crypto = require('node:crypto');
const fs = require('node:fs');
const path = require('node:path');
const {validateNormalizedProjectMetadata} = require('./bootstrap-metadata');
const {validateBootstrapSource} = require('./bootstrap-source');

const MAX_MANIFEST_BYTES = 65536;

function establishedSource(value) {
    return value !== null && typeof value === 'object' && !Array.isArray(value) &&
        Object.keys(value).sort().join(',') === 'evidence,mode' &&
        value.mode === 'ESTABLISHED' && value.evidence === null;
}

function validateProjectManifest({value, coreVersion, allowVersionMigration = false}) {
    const capabilities = Array.isArray(value?.capabilities) ? value.capabilities : [];
    const expected = ['schemaVersion', 'source', 'capabilities', 'project',
        ...(capabilities.length === 0 ? [] : ['capabilityMetadata']),
        'adapter', 'compatibility'];
    if (value === null || typeof value !== 'object' || Array.isArray(value) ||
        Object.keys(value).sort().join(',') !== expected.sort().join(',') ||
        ![1, 2].includes(value.schemaVersion)) throw new Error('project manifest is invalid');
    if (value.schemaVersion === 1) validateBootstrapSource(value.source);
    else if (!establishedSource(value.source)) throw new Error('project manifest is invalid');
    validateNormalizedProjectMetadata({
        capabilities,
        metadata: {
            schemaVersion: 1,
            ...value.project,
            ...(capabilities.length === 0 ? {} : {capabilityMetadata: value.capabilityMetadata}),
        },
    });
    const adapter = value.adapter;
    const adapterValid = adapter === null || (
        adapter !== null && typeof adapter === 'object' && !Array.isArray(adapter) &&
        Object.keys(adapter).sort().join(',') ===
            'bootstrapProtocol,id,packageName,packageVersion' &&
        ['id', 'packageName', 'packageVersion'].every((key) =>
            typeof adapter[key] === 'string' && adapter[key].length > 0
        ) && Number.isSafeInteger(adapter.bootstrapProtocol) &&
        adapter.bootstrapProtocol > 0
    );
    const compatibility = value.compatibility;
    const compatibilityValid = compatibility !== null &&
        typeof compatibility === 'object' && !Array.isArray(compatibility) &&
        Object.keys(compatibility).sort().join(',') ===
            'corePackage,coreVersion,providerProtocol' &&
        compatibility.corePackage === '@kyaulabs/prism-core' &&
        typeof compatibility.coreVersion === 'string' &&
        compatibility.providerProtocol === 1;
    const versionCurrent = compatibilityValid && compatibility.coreVersion === coreVersion;
    if (!adapterValid || !compatibilityValid || (!versionCurrent && !allowVersionMigration)) {
        throw new Error('project manifest is invalid');
    }
    return Object.freeze({value: Object.freeze(value), versionCurrent});
}

function renderProjectManifest({schemaVersion, source, capabilities, metadata, coreVersion, adapter}) {
    const value = {
        schemaVersion,
        source,
        capabilities,
        project: {displayName: metadata.displayName, summary: metadata.summary},
        ...(capabilities.length === 0 ? {} : {capabilityMetadata: metadata.capabilityMetadata}),
        adapter,
        compatibility: {
            corePackage: '@kyaulabs/prism-core',
            coreVersion,
            providerProtocol: 1,
        },
    };
    validateProjectManifest({value, coreVersion});
    return Buffer.from(`${JSON.stringify(value, null, 2)}\n`, 'utf8');
}
