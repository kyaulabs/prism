cmake_minimum_required(VERSION 3.25)
project(sanitizer_detectors LANGUAGES CXX)
foreach(target ubsan_recover asan_error)
  add_executable(${target} ${target}.cpp)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_compile_options(${target} PRIVATE -O0 -g -fno-omit-frame-pointer)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(${target} PRIVATE -no-pie)
  endif()
endforeach()
target_compile_options(ubsan_recover PRIVATE -fsanitize=undefined)
target_link_options(ubsan_recover PRIVATE -fsanitize=undefined)
target_compile_options(asan_error PRIVATE -fsanitize=address)
target_link_options(asan_error PRIVATE -fsanitize=address)
